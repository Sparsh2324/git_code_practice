##  ansible task - Handler to restart nginx
---

-   name:   Manage config file
    hosts:  all
    become: yes

    tasks:
        -   name:   Copy config file
            copy:
                src:    index.html
                dest:   /var/www/html/index.html
            notify: restart Nginx service

    handlers:
        -   name:   restart nginx service
            service:
                name:   nginx
                state:  restarted
